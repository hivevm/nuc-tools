# One case per line, four fields separated by ' | ':
#
#   <pass or fail> | <what the output holds, or -> | <name> | <tree>
#
# A run exits 0 for pass and 1 for fail. The second field is '<key>', then ' @ <subject>' for each
# datum the line names, such as the file and line or the entry in double quotes. A line of the
# output ends in '[<key>]', the rule and a reason word, and holds each subject, in that order and
# with no ASCII letter, digit, hyphen, or underscore beside it; the rest of its wording is free.
#
# A line that starts with '#' is a comment, not a case. A field may start with '!': it then holds
# where no line that ends in '[<key>]' holds the subjects in their order.
#

# that passes the check, not under git, of one file, .devcontainer/compose.yaml:
#
#   services:
#     collector:
#       image: example
#       ports:
#         - "${OBSERVABILITY_BIND:-127.0.0.1}:4318:4318" # OTLP/HTTP
#
# The steps:
#
#   rm <path>              the file or directory removed
#   port [entry]           the entry as one more line '      - [entry]' at the end of compose.yaml
#   line <text>            the text as one more line '        <text>', a key of the long syntax
#   copy <path>            the file of this repository copied over the fixture's
#   write <path> <text>    the file written with exactly the text, each '\n' in it a line break
#                          and each '\r' a carriage return;
#                          the text is all of the step after the path and the one space that
#                          follows it, leading spaces kept
